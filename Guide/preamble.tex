% ======================================================================
%  Math macros --- consistent FEM / continuum-mechanics notation
% ======================================================================
% Vectors & matrices
\renewcommand{\vec}[1]{\bm{#1}}               % vector (bold)
\newcommand{\mat}[1]{\mathbf{#1}}             % matrix (upright bold)
\newcommand{\tensor}[1]{\bm{#1}}             % 2nd-order tensor

% Common differential / vector-calculus operators
\newcommand{\grad}{\nabla}
\newcommand{\divergence}{\nabla\!\cdot\!}
\newcommand{\dd}{\,\mathrm{d}}
\newcommand{\pd}[2]{\frac{\partial #1}{\partial #2}}

% Domains & boundaries
\newcommand{\domain}{\Omega}
\newcommand{\boundary}{\partial\domain}
\newcommand{\GammaD}{\Gamma_{\!D}}            % Dirichlet boundary
\newcommand{\GammaN}{\Gamma_{\!N}}            % Neumann boundary

% Continuum-mechanics quantities
\newcommand{\stress}{\tensor{\sigma}}         % Cauchy stress tensor
\newcommand{\strain}{\tensor{\varepsilon}}    % strain tensor
\newcommand{\disp}{\vec{u}}                    % displacement field
\newcommand{\traction}{\vec{t}}               % surface traction
\newcommand{\bodyforce}{\vec{b}}              % body force

% Voigt-notation column vectors & elasticity matrix
\newcommand{\svec}{\{\sigma\}}                % stress in Voigt form
\newcommand{\evec}{\{\varepsilon\}}          % strain in Voigt form
\newcommand{\Dmat}{\mat{D}}                    % constitutive matrix
\newcommand{\Bmat}{\mat{B}}                    % strain-displacement matrix
\newcommand{\Nmat}{\mat{N}}                    % shape-function matrix
\newcommand{\Jmat}{\mat{J}}                    % Jacobian
\newcommand{\Kmat}{\mat{K}}                    % global stiffness
\newcommand{\Mmat}{\mat{M}}                    % global mass
\newcommand{\ke}{\mat{k}^{e}}                 % element stiffness
\newcommand{\me}{\mat{m}^{e}}                 % element mass

% Discrete DOF / load vectors
\newcommand{\dof}{\vec{d}}                     % nodal DOF vector (global)
\newcommand{\dofe}{\vec{d}^{e}}               % element DOF vector
\newcommand{\force}{\vec{f}}                   % global force vector

% Misc
\newcommand{\transpose}{^{\mathsf{T}}}
\newcommand{\half}{\tfrac{1}{2}}
\DeclareMathOperator{\tr}{tr}
\DeclareMathOperator{\sym}{sym}
